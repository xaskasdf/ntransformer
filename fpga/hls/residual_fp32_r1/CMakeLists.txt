cmake_minimum_required(VERSION 3.24)
project(ntrpu_residual_reference LANGUAGES CXX)
enable_testing()
set(NTRPU_RESIDUAL_FIXTURE "" CACHE FILEPATH "Pinned residual fixture")
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
foreach(kind IN ITEMS residual residual_fuzz)
    add_executable(test_ntrpu_hls_${kind}_r1 ntrpu_residual_fp32_r1.cpp
        "${ROOT}/tests/test_ntrpu_hls_${kind}_r1.cpp")
    target_compile_features(test_ntrpu_hls_${kind}_r1 PRIVATE cxx_std_14)
    target_include_directories(test_ntrpu_hls_${kind}_r1 PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    if(MSVC)
        target_compile_options(test_ntrpu_hls_${kind}_r1 PRIVATE /fp:strict /W4 /wd4068)
    else()
        target_compile_options(test_ntrpu_hls_${kind}_r1 PRIVATE -Wall -Wextra -Wno-unknown-pragmas -ffp-contract=off -fno-fast-math)
    endif()
endforeach()
if(NTRPU_RESIDUAL_FIXTURE)
    add_test(NAME ntrpu_residual_fixture
        COMMAND test_ntrpu_hls_residual_r1 "${NTRPU_RESIDUAL_FIXTURE}")
endif()
add_test(NAME ntrpu_residual_native_fuzz COMMAND test_ntrpu_hls_residual_fuzz_r1)
